 if(!(await indexedDB.databases()).some(x=>x.name===dbName))return null;

  const db=opening.result;if(!db.objectStoreNames.contains(storeName)){db.close();reject(new Error('Fence store missing'));return;}
  const tx=db.transaction(storeName,'readonly');let value=null;const r=tx.objectStore(storeName).get(k);r.onsuccess=()=>{value=r.result??null;};tx.oncomplete=()=>{db.close();resolve(value);};tx.onabort=()=>{db.close();reject(new Error('Native fence read aborted'));};
 };});
